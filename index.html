// DIT FC ONLINE LEAGUE V14.3: Image news content and permission API // Deploy with Supabase CLI: supabase functions deploy dit-news-center import { createClient } from 'https://esm.sh/@supabase/supabase-js@2'; const cors = { 'Access-Control-Allow-Origin':'*', 'Access-Control-Allow-Headers':'authorization, x-client-info, apikey, content-type', 'Access-Control-Allow-Methods':'POST, OPTIONS' }; const respond=(obj:Record,status=200)=>new Response(JSON.stringify(obj),{status,headers:{...cors,'Content-Type':'application/json'}}); const BUCKET='dit-news-media'; const CATEGORIES=new Set(['리그','친선전','선수','공지','기타']); const uuid=(s:unknown)=>typeof s==='string'&&/^[0-9a-f]{8}-[0-9a-f]{4}-[1-8][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/i.test(s); Deno.serve(async(req)=>{ if(req.method==='OPTIONS')return new Response('ok',{headers:cors}); if(req.method!=='POST')return respond({ok:false,error:'POST only'},405); const url=Deno.env.get('SUPABASE_URL'),anon=Deno.env.get('SUPABASE_ANON_KEY'),service=Deno.env.get('SUPABASE_SERVICE_ROLE_KEY'); if(!url||!anon||!service)return respond({ok:false,error:'Missing Supabase server settings'},500); const token=req.headers.get('Authorization')?.replace(/^Bearer\s+/i,'').trim(); if(!token)return respond({ok:false,error:'Login required'},401); const auth=createClient(url,anon,{global:{headers:{Authorization:'Bearer '+token}},auth:{persistSession:false}}); const {data:{user},error:authError}=await auth.auth.getUser(token); if(authError||!user)return respond({ok:false,error:'Valid login required'},401); const db=createClient(url,service,{auth:{persistSession:false}}); try{ const payload=await req.json();const action=String(payload?.action||''); const [{data:a,error:adminError},{data:e,error:editorError}]=await Promise.all([ db.from('league_admins').select('user_id').eq('user_id',user.id).maybeSingle(), db.from('media_news_editors').select('user_id').eq('user_id',user.id).maybeSingle() ]); if(adminError||editorError)return respond({ok:false,error:'Permission lookup failed'},500); const isAdmin=!!a,canEdit=isAdmin||!!e; if(action==='permissions')return respond({ok:true,is_admin:isAdmin,can_edit:canEdit}); if(!canEdit)return respond({ok:false,error:'News editor permission required'},403); if(action==='grant'||action==='revoke'){ if(!isAdmin)return respond({ok:false,error:'Admin permission required'},403); if(!uuid(payload.user_id))return respond({ok:false,error:'Valid Auth UID required'},400); if(action==='grant'){ const {data:target,error:whoError}=await db.auth.admin.getUserById(payload.user_id); if(whoError||!target?.user)return respond({ok:false,error:'Auth user not found'},404); const {error}=await db.from('media_news_editors').upsert({user_id:payload.user_id});if(error)throw error; }else{const {error}=await db.from('media_news_editors').delete().eq('user_id',payload.user_id);if(error)throw error;} return respond({ok:true}); } if(action==='create'){ const title=String(payload.title||'').trim();const category=String(payload.category||'기타'); if(!title||title.length>100||!CATEGORIES.has(category))return respond({ok:false,error:'Title/category invalid'},400); const {data,error}=await db.from('media_news_posts').insert({title,category,created_by:user.id,is_pinned:!!payload.pinned,status:'draft'}).select('id').single(); if(error)throw error;return respond({ok:true,id:data.id}); } if(['upload','publish','delete'].includes(action)){ if(!uuid(payload.id))return respond({ok:false,error:'Invalid post ID'},400); const {data:post,error:findError}=await db.from('media_news_posts').select('id,created_by,status').eq('id',payload.id).maybeSingle(); if(findError)throw findError;if(!post)return respond({ok:false,error:'News not found'},404); if(!isAdmin&&post.created_by!==user.id)return respond({ok:false,error:'Only creator or admin can change this news'},403); if(action==='upload'){ if(post.status!=='draft')return respond({ok:false,error:'Already published'},409); const order=Number(payload.order);if(!Number.isInteger(order)||order<0||order>7)return respond({ok:false,error:'Invalid order'},400); const raw=String(payload.image_data||'');const match=/^data:(image\/(?:jpeg|png|webp));base64,([A-Za-z0-9+/=]+)$/.exec(raw); if(!match)return respond({ok:false,error:'JPG, PNG or WebP required'},400); const mime=match[1],encoded=match[2];if(encoded.length>5592500)return respond({ok:false,error:'Image too large'},413); let bytes:Uint8Array;try{bytes=Uint8Array.from(atob(encoded),c=>c.charCodeAt(0))}catch{return respond({ok:false,error:'Invalid base64 image'},400)} if(bytes.length>4194304||bytes.length<50)return respond({ok:false,error:'Image must be <=4MB'},413); const isJpeg=bytes[0]===0xff&&bytes[1]===0xd8&&bytes[2]===0xff; const isPng=bytes[0]===137&&bytes[1]===80&&bytes[2]===78&&bytes[3]===71; const isWebp=String.fromCharCode(...bytes.slice(0,4))==='RIFF'&&String.fromCharCode(...bytes.slice(8,12))==='WEBP'; if(!((mime==='image/jpeg'&&isJpeg)||(mime==='image/png'&&isPng)||(mime==='image/webp'&&isWebp)))return respond({ok:false,error:'Image type mismatch'},400); const {count,error:countErr}=await db.from('media_news_images').select('*',{count:'exact',head:true}).eq('post_id',post.id);if(countErr)throw countErr; if((count||0)>=8)return respond({ok:false,error:'Max 8 images'},400); const ext=mime==='image/jpeg'?'jpg':mime==='image/png'?'png':'webp';const path=`${post.id}/${order}-${crypto.randomUUID()}.${ext}`; const {error:storageError}=await db.storage.from(BUCKET).upload(path,bytes,{contentType:mime,upsert:false,cacheControl:'3600'}); if(storageError)throw storageError; const {error:saveError}=await db.from('media_news_images').insert({post_id:post.id,path,sort_order:order}); if(saveError){await db.storage.from(BUCKET).remove([path]);throw saveError;} return respond({ok:true}); } if(action==='publish'){ const {count,error}=await db.from('media_news_images').select('*',{head:true,count:'exact'}).eq('post_id',post.id); if(error)throw error;if(!count)return respond({ok:false,error:'At least one image is required'},400); const {error:updateError}=await db.from('media_news_posts').update({status:'published',published_at:new Date().toISOString()}).eq('id',post.id);if(updateError)throw updateError; return respond({ok:true}); } if(action==='delete'){ const {data:images,error}=await db.from('media_news_images').select('path').eq('post_id',post.id);if(error)throw error; // Database first. If file cleanup fails, inaccessible files can be cleaned later by the operator. const {error:deletion}=await db.from('media_news_posts').delete().eq('id',post.id);if(deletion)throw deletion; if(images?.length){const {error:cleanup}=await db.storage.from(BUCKET).remove(images.map(x=>x.path));if(cleanup)console.error('Image cleanup required:',cleanup)} return respond({ok:true}); } } return respond({ok:false,error:'Unknown action'},400); }catch(error){console.error('dit-news-center',error);return respond({ok:false,error:'Server error; check function logs'},500)} });
